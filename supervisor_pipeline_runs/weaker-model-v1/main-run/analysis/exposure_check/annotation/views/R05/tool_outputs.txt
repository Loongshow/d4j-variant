   481	    /**
   482	     * Adds a data item to the series and sends a
   483	     * {@link org.jfree.data.general.SeriesChangeEvent} to all registered
   484	     * listeners.
   485	     *
   486	     * @param item  the (timeperiod, value) pair (<code>null</code> not
   487	     *              permitted).
   488	     * @param notify  notify listeners?
   489	     */
   490	    public void add(TimeSeriesDataItem item, boolean notify) {
   491	        if (item == null) {
   492	            throw new IllegalArgumentException("Null 'item' argument.");
   493	        }
   494	        if (!item.getPeriod().getClass().equals(this.timePeriodClass)) {
   495	            StringBuffer b = new StringBuffer();
   496	            b.append("You are trying to add data where the time period class ");
   497	            b.append("is ");
   498	            b.append(item.getPeriod().getClass().getName());
   499	            b.append(", but the TimeSeries is expecting an instance of ");
   500	            b.append(this.timePeriodClass.getName());
   501	            b.append(".");
   502	            throw new SeriesException(b.toString());
   503	        }
   504	
   505	        // make the change (if it's not a duplicate time period)...
   506	        boolean added = false;
   507	        int count = getItemCount();
   508	        if (count == 0) {
   509	            this.data.add(item);
   510	            added = true;
   511	        }
   512	        else {
   513	            RegularTimePeriod last = getTimePeriod(getItemCount() - 1);
   514	            if (item.getPeriod().compareTo(last) > 0) {
   515	                this.data.add(item);
   516	                added = true;
   517	            }
   518	            else {
   519	                int index = Collections.binarySearch(this.data, item);
   520	                if (index < 0) {
   521	                    this.data.add(-index - 1, item);
   522	                    added = true;
   523	                }
   524	                else {
   525	                    StringBuffer b = new StringBuffer();
   526	                    b.append("You are attempting to add an observation for ");
   527	                    b.append("the time period ");
   528	                    b.append(item.getPeriod().toString());
   529	                    b.append(" but the series already contains an observation");
   530	                    b.append(" for that time period. Duplicates are not ");
   531	                    b.append("permitted.  Try using the addOrUpdate() method.");
   532	                    throw new SeriesException(b.toString());
   533	                }
   534	            }
   535	        }
   536	        if (added) {
   537	            // check if this addition will exceed the maximum item count...
   538	            if (getItemCount() > this.maximumItemCount) {
   539	                this.data.remove(0);
   540	            }
   541	
   542	            removeAgedItems(false);  // remove old items if necessary, but
   543	                                     // don't notify anyone, because that
   544	                                     // happens next anyway...
   545	            if (notify) {
   546	                fireSeriesChanged();
   547	            }
   548	        }
   549	
   550	    }
