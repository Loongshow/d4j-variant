   346	    /**
   347	     Wrap the supplied HTML around this element.
   348	     @param html HTML to wrap around this element, e.g. {@code <div class="head"></div>}. Can be arbitralily deep.
   349	     @return this element, for chaining.
   350	     */
   351	    public Element wrap(String html) {
   352	        Validate.notEmpty(html);
   353	
   354	        Element wrapBody = Parser.parseBodyFragmentRelaxed(html, baseUri).body();
   355	        Elements wrapChildren = wrapBody.children();
   356	        Element wrap = wrapChildren.first();
   357	        if (wrap == null) // nothing to wrap with; noop
   358	            return null;
   359	
   360	        Element deepest = getDeepChild(wrap);
   361	        parentNode.replaceChild(this, wrap);
   362	        deepest.addChildren(this);
   363	
   364	        // remainder (unbalananced wrap, like <div></div><p></p> -- The <p> is remainder
   365	        if (wrapChildren.size() > 1) {
   366	            for (int i = 1; i < wrapChildren.size(); i++) { // skip first
   367	                Element remainder = wrapChildren.get(i);
   368	                remainder.parentNode.removeChild(remainder);
   369	                wrap.appendChild(remainder);
   370	            }
   371	        }
   372	        return this;
   373	    }

   454	        Validate.notNull(search);

   457	        for (int i = 0; i < elements.size(); i++) {

   462	        return null;
