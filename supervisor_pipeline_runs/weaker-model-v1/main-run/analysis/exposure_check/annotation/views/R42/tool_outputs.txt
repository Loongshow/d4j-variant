   554	        if ((u == 0) || (v == 0)) {
   555	            if ((u == Integer.MIN_VALUE) || (v == Integer.MIN_VALUE)) {
   556	                throw new ArithmeticException("overflow: gcd is 2^31");
   557	            }
   558	            return Math.abs(u) + Math.abs(v);
   559	        }
   560	        //if either operand is abs 1, return 1:
   561	        if (Math.abs(u) == 1 || Math.abs(v) == 1) {
   562	            return 1;
   563	        }
   564	        // keep u and v negative, as negative integers range down to
   565	        // -2^31, while positive numbers can only be as large as 2^31-1
   566	        // (i.e. we can't necessarily negate a negative number without

   572	        while ((u&1)==0 && (v&1)==0 && k<31) { // while u and v are both even...
   573	            u/=2; v/=2; k++; // cast out twos.
   574	        }
   575	        if (k==31) {
   576	            throw new ArithmeticException("overflow: gcd is 2^31");
   577	        }

   579	        //     one is odd.

   581	        // t negative: u was odd, v may be even (t replaces v)
   582	        // t positive: u was even, v is odd (t replaces u)
   583	        do {
   584	            /* assert u<0 && v<0; */
   585	            // B4/B3: cast out twos from t.
   586	            while ((t&1)==0) { // while t is even..
   587	                t/=2; // cast out twos
   588	            }
   589	            // B5 [reset max(u,v)]

   592	            } else {
   593	                v = t;
   594	            }
   595	            // B6/B3. at this point both u and v should be odd.
   596	            t = (v - u)/2;
   597	            // |u| larger: t positive (replace u)
   598	            // |v| larger: t negative (replace v)
   599	        } while (t!=0);
   600	        return -u*(1<<k); // gcd is u*2^k
