     * a comma, newline or double quote
     * @throws IOException if error occurs on underlying Writer

    public static void escapeCsv(Writer out, String str) throws IOException {
        if (StringUtils.containsNone(str, CSV_SEARCH_CHARS)) {
            if (str != null) {
                out.write(str);
            }
            return;
        }
        out.write(CSV_QUOTE);
        for (int i = 0; i < str.length(); i++) {
            char c = str.charAt(i);
            if (c == CSV_QUOTE) {
                out.write(CSV_QUOTE); // escape double quote
            }
            out.write(c);
        }
        out.write(CSV_QUOTE);
    }

     * @param out Writer to write the input String to, with enclosing double quotes 
     * removed and embedded double quotes unescaped, <code>null</code> if null string input
     * @throws IOException if error occurs on underlying Writer
     * @since 2.4
     */
    public static void unescapeCsv(Writer out, String str) throws IOException {
        if (str == null) {
            return;
        }
        if (str.length() < 2) {
            out.write(str);
            return;
        }
        if ( str.charAt(0) != CSV_QUOTE || str.charAt(str.length() - 1) != CSV_QUOTE ) {
            out.write(str);
            return;
        }

        // strip quotes
        String quoteless = str.substring(1, str.length() - 1);

        if ( StringUtils.containsAny(quoteless, CSV_SEARCH_CHARS) ) {
            // deal with escaped quotes; ie) ""
            str = StringUtils.replace(quoteless, CSV_QUOTE_STR + CSV_QUOTE_STR, CSV_QUOTE_STR);
        }

        out.write(str);
    }

            return text;

        if (end == -1) {

            if (--max == 0) {

            }

        }

   745	     * </p>
   746	     *
   747	     * see <a href="http://en.wikipedia.org/wiki/Comma-separated_values">Wikipedia</a> and
   748	     * <a href="http://tools.ietf.org/html/rfc4180">RFC 4180</a>.
   749	     *
   750	     * @param str the input CSV column String, may be null
   751	     * @param out Writer to write input string to, enclosed in double quotes if it contains
   752	     * a comma, newline or double quote
   753	     * @throws IOException if error occurs on underlying Writer
   754	     * @since 2.4
   755	     */
   756	    public static void escapeCsv(Writer out, String str) throws IOException {
   757	        if (StringUtils.containsNone(str, CSV_SEARCH_CHARS)) {
   758	            if (str != null) {
   759	                out.write(str);
   760	            }
   761	            return;
   762	        }
   763	        out.write(CSV_QUOTE);
   764	        for (int i = 0; i < str.length(); i++) {
   765	            char c = str.charAt(i);
   766	            if (c == CSV_QUOTE) {
   767	                out.write(CSV_QUOTE); // escape double quote
   768	            }
   769	            out.write(c);
   770	        }
   771	        out.write(CSV_QUOTE);
   772	    }

   774	    /**
   775	     * <p>Returns a <code>String</code> value for an unescaped CSV column. </p>
  1938	     * @param value  the value to write as a field, not null
  1939	     * @param searchChars  the delimiter and the line break characters,
  1940	     *  not null
  1941	     * @return the field for the value
  1942	     */
  1943	    public static String toField(String value, String searchChars) {
  1944	        StringBuffer buf = new StringBuffer(value.length() + 2);
  1945	        boolean enclose = false;
  1946	        for (int i = 0; i < value.length(); i++) {
  1947	            char ch = value.charAt(i);
  1948	            if (ch == '"') {
  1949	                buf.append(ch);
  1950	            } else if (searchChars.indexOf(ch) >= 0) {
  1951	                enclose = true;
  1952	            }
  1953	            buf.append(ch);
  1954	        }
  1955	        if (enclose) {
  1956	            return "\"" + buf.toString() + "\"";
  1957	        }
  1958	        return value;
  1959	    }
